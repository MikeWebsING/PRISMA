    public static char[] traducir(char[] codigo) {

            ' ',' ',' ',' ','M','o','v',' ','A','h',',',' ','0','9','h','\n',

                    char[] saltoLinea = {'\n'};

                    cuerpo = concatenar(cuerpo, saltoLinea);

                    char[] saltoLinea = {'\n'};
                    char[] dosPuntos = {':','\n'};
                    
                    cuerpo = concatenar(cuerpo, jmpAsm);
                    cuerpo = concatenar(cuerpo, etiquetaInicio);
                    cuerpo = concatenar(cuerpo, saltoLinea);
                    cuerpo = concatenar(cuerpo, etiquetaFin);
                    cuerpo = concatenar(cuerpo, dosPuntos);
                } else {

                    char[] lineaAsm = traducirLinea(linea);

    public static char[] concatenar(char[] arregloUno, char[] arregloDos) {
        char[] resultado = new char[arregloUno.length + arregloDos.length];
        for (int i = 0; i < arregloUno.length; i++) {
            resultado[i] = arregloUno[i];
        }
        for (int i = 0; i < arregloDos.length; i++) {
            resultado[arregloUno.length + i] = arregloDos[i];
        }
        return resultado;
    }

    public static boolean coincideExactamente(char[] fuente, int inicio, int longitudLinea, char[] busqueda) {
        if (longitudLinea != busqueda.length) {
            return false;
        }
        for (int i = 0; i < longitudLinea; i++) {
            if (fuente[inicio + i] != busqueda[i]) {
                return false;
            }
        }
        return true;
    }

    public static boolean empiezaCon(char[] fuente, int inicio, int longitudLinea, char[] busqueda) {
        if (longitudLinea < busqueda.length) {
            return false;
        }
        for (int i = 0; i < busqueda.length; i++) {
            if (fuente[inicio + i] != busqueda[i]) {
                return false;
            }
        }
        return true;
    }

    public static int encontrarCaracter(char[] fuente, int inicio, int fin, char busqueda) {
        for (int i = inicio; i < fin; i++) {
            if (fuente[i] == busqueda) {
                return i;
            }
        }
        return -1;
    }

    public static int encontrarCaracterDesdeAtras(char[] fuente, int inicio, int fin, char busqueda) {
        for (int i = fin - 1; i >= inicio; i--) {
            if (fuente[i] == busqueda) {
                return i;
            }
        }
        return -1;
    }

    public static boolean contieneCaracter(char[] fuente, int inicio, int fin, char busqueda) {
        return encontrarCaracter(fuente, inicio, fin, busqueda) != -1;
    }

    public static char[] extraerSubcadena(char[] fuente, int inicio, int fin) {
        int tamano = fin - inicio;
        if (tamano <= 0) {
            return new char[0];
        }
        char[] resultado = new char[tamano];
        for (int i = 0; i < tamano; i++) {
            resultado[i] = fuente[inicio + i];
        }
        return resultado;
    }

    public static boolean esNumero(char[] s) {
        if (s.length == 0) return false;
        for (int i = 0; i < s.length; i++) {
            if (s[i] < '0' || s[i] > '9') return false;
        }
        return true;
    }

    public static boolean esCadena(char[] s) {
        if (s.length < 2) return false;
        return s[0] == '"' && s[s.length - 1] == '"';
    }

            char[] tmp = {'a','d','d',' ','A','l',',',' ','C','l','\n','a','d','d',' ','A','l',',',' ','3','0','h','\n'};
            opAsm = tmp;
        } else if (operador == '-') {

            opAsm = tmp;
        } else if (operador == '/') {
            char[] tmp = {'m','o','v',' ','A','h',',',' ','0','\n','d','i','v',' ','C','l','\n','a','d','d',' ','A','l',',',' ','3','0','h','\n'};
            opAsm = tmp;
        } else if (operador == '%') {
            char[] tmp = {'m','o','v',' ','A','h',',',' ','0','\n','d','i','v',' ','C','l','\n','a','d','d',' ','A','h',',',' ','3','0','h','\n','m','o','v',' ','A','l',',',' ','A','h','\n'};
            opAsm = tmp;

        char[] p4 = {'m','o','v',' ','S','i',',',' ','o','f','f','s','e','t',' '};

    public static char[] generarNombreEtiqueta() {
        char[] nombre = {'E','t','i','q','u','e','t','a','0','0'};
        int n = contadorEtiquetas++;
        nombre[8] = (char)('0' + (n / 10));
        nombre[9] = (char)('0' + (n % 10));
        return nombre;
    }

    public static char[] generarComparacion(char[] condicion, char[] etiquetaSalto) {
        // Simplificado: id op id o id op num
        int idxOp = -1;
        char[] op = new char[0];
        
        char[][] ops = {{'=','='}, {'!','='}, {'>','='}, {'<','='}, {'>'}, {'<'}};
        for (int i = 0; i < ops.length; i++) {
            int pos = encontrarSubcadena(condicion, ops[i]);
            if (pos != -1) {
                idxOp = pos;
                op = ops[i];
                break;
            }
        }
        
        if (idxOp == -1) return new char[0];
        
        char[] var1 = extraerSubcadena(condicion, 0, idxOp);
        char[] var2 = extraerSubcadena(condicion, idxOp + op.length, condicion.length);
        
        // Limpiar espacios
        var1 = limpiarEspacios(var1);
        var2 = limpiarEspacios(var2);
        
        char[] c1;
        if (esNumero(var1)) {
            char[] p1 = {'m','o','v',' ','A','l',',','\''};
            char[] p2 = {'\'','\n'};
            c1 = concatenar(p1, var1);
            c1 = concatenar(c1, p2);
        } else {
            char[] p1 = {'m','o','v',' ','S','i',',',' ','o','f','f','s','e','t',' '};
            char[] p2 = {'+','2','\n','m','o','v',' ','A','l',',',' ','b','y','t','e',' ','p','t','r',' ','[','S','i',']','\n'};
            c1 = concatenar(p1, var1);
            c1 = concatenar(c1, p2);
        }
        
        char[] c2;
        if (esNumero(var2)) {
            char[] p1 = {'m','o','v',' ','B','l',',','\''};
            char[] p2 = {'\'','\n'};
            c2 = concatenar(p1, var2);
            c2 = concatenar(c2, p2);
        } else {
            char[] p1 = {'m','o','v',' ','D','i',',',' ','o','f','f','s','e','t',' '};
            char[] p2 = {'+','2','\n','m','o','v',' ','B','l',',',' ','b','y','t','e',' ','p','t','r',' ','[','D','i',']','\n'};
            c2 = concatenar(p1, var2);
            c2 = concatenar(c2, p2);
        }
        
        char[] p3 = {'c','m','p',' ','A','l',',',' ','B','l','\n'};
        char[] jmp;
        
        if (op.length == 2) {
            if (op[0] == '=' && op[1] == '=') jmp = new char[]{'j','n','e',' '};
            else if (op[0] == '!' && op[1] == '=') jmp = new char[]{'j','e',' '};
            else if (op[0] == '>' && op[1] == '=') jmp = new char[]{'j','l',' '};
            else if (op[0] == '<' && op[1] == '=') jmp = new char[]{'j','g',' '};
            else jmp = new char[]{'j','n','e',' '};
        } else {
            if (op[0] == '>') jmp = new char[]{'j','l','e',' '};
            else if (op[0] == '<') jmp = new char[]{'j','g','e',' '};
            else jmp = new char[]{'j','n','e',' '};
        }
        
        char[] r = concatenar(c1, c2);
        r = concatenar(r, p3);
        r = concatenar(r, jmp);
        r = concatenar(r, etiquetaSalto);
        r = concatenar(r, new char[]{'\n'});
        
        return r;
    }

    public static int encontrarSubcadena(char[] fuente, char[] busqueda) {
        for (int i = 0; i <= fuente.length - busqueda.length; i++) {
            boolean coincide = true;
            for (int j = 0; j < busqueda.length; j++) {
                if (fuente[i + j] != busqueda[j]) {
                    coincide = false;
                    break;
                }
            }
            if (coincide) return i;
        }
        return -1;
    }

    public static char[] limpiarEspacios(char[] s) {
        int inicio = 0;
        while (inicio < s.length && (s[inicio] == ' ' || s[inicio] == '\t')) inicio++;
        int fin = s.length;
        while (fin > inicio && (s[fin - 1] == ' ' || s[fin - 1] == '\t')) fin--;
        return extraerSubcadena(s, inicio, fin);
    }

    public static char[] traducirLinea(char[] linea) {
        int indiceIgual = encontrarCaracter(linea, 0, linea.length, '=');
        if (indiceIgual != -1) {
            int inicioId = 0;
            while (inicioId < indiceIgual && (linea[inicioId] == ' ' || linea[inicioId] == '\t')) inicioId++;
            int finId = indiceIgual;
            while (finId > inicioId && (linea[finId - 1] == ' ' || linea[finId - 1] == '\t')) finId--;
            char[] destino = extraerSubcadena(linea, inicioId, finId);
            
            int inicioDer = indiceIgual + 1;
            while (inicioDer < linea.length && (linea[inicioDer] == ' ' || linea[inicioDer] == '\t')) inicioDer++;
            int finDer = linea.length;
            while (finDer > inicioDer && (linea[finDer - 1] == ' ' || linea[finDer - 1] == '\t')) finDer--;
            
            int indiceSuma = encontrarCaracter(linea, inicioDer, finDer, '+');
            int indiceResta = encontrarCaracter(linea, inicioDer, finDer, '-');
            int indiceMult = encontrarCaracter(linea, inicioDer, finDer, '*');
            int indiceDiv = encontrarCaracter(linea, inicioDer, finDer, '/');
            int indiceMod = encontrarCaracter(linea, inicioDer, finDer, '%');
            
            int indiceOperador = -1;
            char operador = ' ';
            if (indiceSuma != -1) { indiceOperador = indiceSuma; operador = '+'; }
            else if (indiceResta != -1) { indiceOperador = indiceResta; operador = '-'; }
            else if (indiceMult != -1) { indiceOperador = indiceMult; operador = '*'; }
            else if (indiceDiv != -1) { indiceOperador = indiceDiv; operador = '/'; }
            else if (indiceMod != -1) { indiceOperador = indiceMod; operador = '%'; }
            
            if (indiceOperador != -1) {
                int finVar1 = indiceOperador;
                while (finVar1 > inicioDer && (linea[finVar1 - 1] == ' ' || linea[finVar1 - 1] == '\t')) finVar1--;
                char[] var1 = extraerSubcadena(linea, inicioDer, finVar1);
                
                int inicioVar2 = indiceOperador + 1;
                while (inicioVar2 < finDer && (linea[inicioVar2] == ' ' || linea[inicioVar2] == '\t')) inicioVar2++;
                char[] var2 = extraerSubcadena(linea, inicioVar2, finDer);
                
                return generarAritmetica(destino, var1, var2, operador);
            } else {
                char[] valor = extraerSubcadena(linea, inicioDer, finDer);
                return generarAsignacion(destino, valor);
            }
        }
        return new char[0];
    }
