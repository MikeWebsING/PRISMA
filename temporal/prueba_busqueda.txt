MODULO SistemaBusqueda
VARIABLES
    ENTERO objetivo = 20
    ENTERO posicion = -1
    ENTERO nActual = 0
    ENTERO n1 = 3
    ENTERO n2 = -2
    ENTERO n3 = 8
    ENTERO n4 = 15
    ENTERO n5 = 20
    ENTERO n6 = 7
    ENTERO contador = 0
    ENTERO aux = 0
    ENTERO resultadoBusqueda = -1
FIN-VARIABLES

FUNCION ENTERO procesarNumero()
    SI (nActual > 0) ENTONCES
        aux = nActual % 2
        SI (aux == 0) ENTONCES
            SI (nActual == objetivo) ENTONCES
                RETORNA contador
            FIN-SI
        FIN-SI
    FIN-SI
    RETORNA -1
FIN-FUNCION

PRINCIPAL
    PARA (contador = 0, contador < 6, contador = contador + 1)
        SI (contador == 0) ENTONCES
            nActual = n1
        FIN-SI
        SI (contador == 1) ENTONCES
            nActual = n2
        FIN-SI
        SI (contador == 2) ENTONCES
        nActual = n3
        FIN-SI
        SI (contador == 3) ENTONCES
            nActual = n4
        FIN-SI
        SI (contador == 4) ENTONCES
            nActual = n5
        FIN-SI
        SI (contador == 5) ENTONCES
            nActual = n6
        FIN-SI

        resultadoBusqueda = procesarNumero()
        SI (resultadoBusqueda != -1) ENTONCES
            posicion = resultadoBusqueda
        FIN-SI
    FIN-PARA

    IMPRIME("Listado de numeros:")
    PARA (contador = 0, contador < 6, contador = contador + 1)
        SI (contador == 0) ENTONCES
            IMPRIME(n1)
        FIN-SI
        SI (contador == 1) ENTONCES
            IMPRIME(n2)
        FIN-SI
        SI (contador == 2) ENTONCES
            IMPRIME(n3)
        FIN-SI
        SI (contador == 3) ENTONCES
            IMPRIME(n4)
        FIN-SI
        SI (contador == 4) ENTONCES
            IMPRIME(n5)
        FIN-SI
        SI (contador == 5) ENTONCES
            IMPRIME(n6)
        FIN-SI
    FIN-PARA

    IMPRIME("Posicion encontrada:")
    IMPRIME(posicion)
FIN-PRINCIPAL
FIN-MODULO
