MODULO ej_num
VARIABLES
    ENTERO num = 0
    ENTERO seguir = 1
    ENTERO i = 0
FIN-VARIABLES

PRINCIPAL
    MIENTRAS ( seguir == 1 )
        IMPRIME ( "Ingrese un numero entre 0 y 9:" )
        LEER ( num )

        SI ( num == 0 O num == 9 ) ENTONCES
        SINO
            SI ( num % 2 == 0 ) ENTONCES
                IMPRIME ( "Numeros pares:" )
                PARA ( i = 0 , i <= 9 , i = i + 2 )
                    IMPRIME ( i )
                FIN-PARA
            SINO
                IMPRIME ( "Numeros impares:" )
                i = 1
                MIENTRAS ( i <= 9 )
                    IMPRIME ( i )
                    i = i + 2
                FIN-MIENTRAS
            FIN-SI
        FIN-SI

        IMPRIME ( "Ejecutar de nuevo? (1=Si, 0=No)" )
        LEER ( seguir )
    FIN-MIENTRAS
FIN-PRINCIPAL
FIN-MODULO
