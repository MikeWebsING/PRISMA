MODULO CalculoPoligonos

VARIABLES
  DECIMAL resultado
FIN-VARIABLES

FUNCION DECIMAL calcularArea(TEXTO tipo, DECIMAL base, DECIMAL altura)
  DECIMAL area
  
  SI (tipo == "TRIANGULO") ENTONCES
    area = (base * altura) / 2.0
  FIN-SI
  
  SI (tipo == "CUADRADO") ENTONCES
    area = base * base
  FIN-SI
  
  SI (tipo == "RECTANGULO") ENTONCES
    area = base * altura
  FIN-SI
  
  RETORNA area
FIN-FUNCION

PRINCIPAL
  IMPRIME("--- CALCULADORA DE AREAS ---")
  
  IMPRIME("Area del Triangulo (base 10, altura 5):")
  resultado = calcularArea("TRIANGULO", 10.0, 5.0)
  IMPRIME(resultado)
  
  IMPRIME("Area del Cuadrado (lado 4):")
  resultado = calcularArea("CUADRADO", 4.0, 4.0)
  IMPRIME(resultado)
  
  IMPRIME("Area del Rectangulo (base 8, altura 3):")
  resultado = calcularArea("RECTANGULO", 8.0, 3.0)
  IMPRIME(resultado)
FIN-PRINCIPAL

FIN-MODULO
