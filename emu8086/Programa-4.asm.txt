; Imprime mensaje con el servicio 09h
IMPRIME Macro Mensaje
    Mov Ah, 09h
    mov dx, offset Mensaje
    int 21h  
EndM 

POSICIONA Macro Cadena, Linea  
    mov Dx, offset Cadena
    mov Si, Dx
    add Si, 1 ; Movemos Dx a la direccion del byte 1
    
    mov Ah, byte ptr (Si)   
    mov Col, 80
    sub Col, Ah

    xor Ax, Ax  ;Pone a cero Ax o sea Ax=0
    mov Al, Col ;Pasamos valor de Col a Ax

    mov Bl, 2   ;Al contiene el resultado de la division o sea la columna donde quiero imprimir
    div Bl      
    mov Col, Al ;Col contiene la columna
   

    mov dh, Linea
    mov dl, Col
    mov bh, 0
    mov ah, 2
    int 10h     
    
EndM 

LEER Macro Entrada
    mov Ah, 0Ah 
    mov Dx, offset Entrada ;Dx contiene la direccion de la variable Nom
    int 21h    
EndM 


CLS Macro
    Mov Ah, 0Fh
    int 10h
    Mov Ah, 0h
    int 10h
EndM


.MODEL SMALL   ; Define el tamano de memoria   

.CODE          ; indica el inicio del programa

Inicio:        ; Etiqueta de inicio obligatorio 


mov Ax, @Data   ;asignamos direccion   
mov Ds, Ax      ;al sagmento de datos (DS) 


IMPRIME Mge1
LEER A           
IMPRIME Salto
IMPRIME Mge2
LEER B
IMPRIME Salto

  
xor Ax, Ax
mov Si, offset A+2
mov Ah, byte ptr [Si]
mov Di, offset B+2
mov Al, byte ptr [Di] 
cmp Ah, Al 

ja MayorA
jb MayorB
IMPRIME Mge5
jmp Fin 

MayorA:
IMPRIME Mge3
jmp Fin

MayorB:
IMPRIME Mge4


Fin:                                                                 
mov ax, 4C00h
int 21h      


.DATA  
                                                                
Salto db 10,13,24h
 
Mge1 db "Valor de A:  $" 
Mge2 db "Valor de B:  $"
Mge3 db "El mayor es A  $"
Mge4 db "El mayor es B  $"
Mge5 db "Son iguales $"


A db 5,?,5 dup (24h)
B db 5,?,5 dup (24h)
Col db 2 dup (24h)


.STACK
END Inicio
