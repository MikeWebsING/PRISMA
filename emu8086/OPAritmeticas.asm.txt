; Imprime mensaje con el servicio 09h
IMPRIME Macro Mensaje
    Mov Ah, 09h
    mov dx, offset Mensaje
    int 21h  
EndM  

LEER Macro Entrada
    mov Ah, 0Ah 
    mov Dx, offset Entrada ;Dx contiene la direccion de la variable Nom
    int 21h    
EndM 

CLS Macro
    Mov Ah, 0Fh
    int 10h
    Mov Ah, 0h
    int 10h
EndM
